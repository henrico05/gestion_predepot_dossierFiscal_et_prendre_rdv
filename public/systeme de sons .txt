import { Howl } from 'howler'

const sounds = {
	success: new Howl({
		src: ['/sounds/success.mp3']
	}),
	
	error: new Howl({
		src: ['/sounds/error.mp3']
	}),
	
	notification: new Howl({
		src: ['/sounds/notification.mp3']
	}),
	
	delete: new Howl({
		src: ['/sounds/delete.mp3']
	}),
	
	update: new Howl({
		src: ['/sounds/update.mp3']
	}),
	
	connection: new Howl({
		src: ['/sounds/connection.mp3']
	}),
	
	disconnection: new Howl({
		src: ['/sounds/disconnection.mp3']
	})
}

export const playSound = (type) => {
	if (sounds[type]) {
		sounds[type].play()
	}
}
